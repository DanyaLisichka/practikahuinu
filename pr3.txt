const http = require('http');

const server = http.createServer((req, res) => {
    res.writeHead(200, {
        'Content-Type': 'text/plain; charset=utf-8'
    });

    res.end('Node.js приложение работает!');
});

server.listen(3000, '0.0.0.0', () => {
    console.log('Server started on port 3000');
});
