const http = require('http');
const os = require('os');
const { Pool } = require('pg');
const { createClient } = require('redis');

const pool = new Pool({
    host: process.env.DB_HOST || 'db',
    port: 5432,
    user: 'app',
    password: process.env.DB_PASSWORD,
    database: 'appdb'
});

const redis = createClient({
    url: `redis://${process.env.REDIS_HOST || 'cache'}:6379`
});

redis.on('error', console.error);

async function start() {
    await redis.connect();
    await pool.query('SELECT 1');

    const server = http.createServer(async (req, res) => {
        try {
            const db = await pool.query('SELECT NOW() AS time');
            const requests = await redis.incr('request_count');

            res.writeHead(200, {
                'Content-Type': 'application/json; charset=utf-8'
            });

            res.end(JSON.stringify({
                message: 'Docker Compose работает!',
                database: db.rows[0].time,
                redisRequests: requests,
                host: os.hostname()
            }));
        } catch (error) {
            console.error(error);
            res.writeHead(500);
            res.end('Internal Server Error');
        }
    });

    server.listen(3000, '0.0.0.0', () => {
        console.log('Node.js server started on port 3000');
    });
}

start().catch(error => {
    console.error(error);
    process.exit(1);
});
